  @override
  Future<List<domain.Intake>> getToday() async {
    final now = DateTime.now();
    final startOfDay = DateTime(now.year, now.month, now.day);
    final endOfDay = startOfDay.add(const Duration(days: 1));

    final rows = await (_db.select(_db.intakes)
      ..where((t) =>
      t.plannedAt.isBiggerOrEqualValue(startOfDay) &
      t.plannedAt.isSmallerThanValue(endOfDay))
      ..orderBy([(t) => OrderingTerm(expression: t.plannedAt)]))
        .get();

    return rows.map(_toDomain).toList();
  }

  @override
  Future<List<domain.Intake>> getByMedicationId(int medicationId) async {
    final rows = await (_db.select(_db.intakes)
      ..where((t) => t.medicationId.equals(medicationId))
      ..orderBy([(t) => OrderingTerm(expression: t.plannedAt, mode: OrderingMode.desc)]))
        .get();
    return rows.map(_toDomain).toList();
  }

  @override
  Future<int> add(domain.Intake intake) async {
    return _db.into(_db.intakes).insert(_toCompanion(intake));
  }

/// - daily: РїСЂРёС‘РјС‹ РєР°Р¶РґС‹Р№ РґРµРЅСЊ
/// - everyOtherDay: РїСЂРёС‘РјС‹ С‡РµСЂРµР· РґРµРЅСЊ (СЃС‡РёС‚Р°РµРј РѕС‚ startDate)
/// - specificDaysOfWeek: С‚РѕР»СЊРєРѕ РІ СѓРєР°Р·Р°РЅРЅС‹Рµ РґРЅРё РЅРµРґРµР»Рё
List<Intake> generateIntakesForDate({
  required Schedule schedule,
  required DateTime date,
  required DateTime scheduleStartDate,
}) {
  final intakes = <Intake>[];

  // РџСЂРѕРІРµСЂСЏРµРј, РїРѕРїР°РґР°РµС‚ Р»Рё РґР°С‚Р° РІ СЂР°СЃРїРёСЃР°РЅРёРµ
  if (!_shouldTakeOnDate(schedule, date, scheduleStartDate)) {
    return intakes;
  }

  // РЎРѕР·РґР°С‘Рј Intake РґР»СЏ РєР°Р¶РґРѕРіРѕ РІСЂРµРјРµРЅРё
  for (final minutes in schedule.timesInMinutes) {
    final plannedAt = DateTime(
      date.year,
      date.month,
      date.day,
      minutes ~/ 60,
      minutes % 60,
    );
    intakes.add(Intake(
      scheduleId: schedule.id!,
      medicationId: schedule.medicationId,
      plannedAt: plannedAt,
    ));
  }

  return intakes;
}

bool _shouldTakeOnDate(
    Schedule schedule,
    DateTime date,
    DateTime startDate,
    ) {
  switch (schedule.periodicity) {
    case SchedulePeriodicity.daily:
      return true;

    case SchedulePeriodicity.everyOtherDay:
      final start = DateTime(startDate.year, startDate.month, startDate.day);
      final current = DateTime(date.year, date.month, date.day);
      final diff = current.difference(start).inDays;
      return diff >= 0 && diff % 2 == 0;

    case SchedulePeriodicity.specificDaysOfWeek:
      if (schedule.weekdays == null || schedule.weekdays!.isEmpty) {
        return false;
      }
      return schedule.weekdays!.contains(date.weekday);
  }
}

      schedule: savedSchedule,
      date: day,
      scheduleStartDate: startDate,
    );

    }

                  style: const TextStyle(fontSize: 14, color: Colors.black54),
                ),
              ],
            ),
          ),
          _buildStatusBadge(item.status),
        ],
      ),
    );
  }

  Widget _buildStatusBadge(IntakeStatus status) {
    Color color;
    String text;
    switch (status) {
      case IntakeStatus.taken:
        color = AppTheme.statusTaken;
        text = 'РџСЂРёРЅСЏС‚Рѕ';
      case IntakeStatus.skipped:
        color = AppTheme.statusSkipped;
        text = 'РџСЂРѕРїСѓС‰РµРЅРѕ';
      case IntakeStatus.postponed:
        color = AppTheme.statusPostponed;
        text = 'РћС‚Р»РѕР¶РµРЅРѕ';
      case IntakeStatus.pending:
        color = AppTheme.statusPending;
        text = 'РћР¶РёРґР°РµС‚СЃСЏ';
    }

    return Container(
      padding: const EdgeInsets.symmetric(horizontal: 10, vertical: 4),
      decoration: BoxDecoration(
        color: color.withValues(alpha: 0.15),
        borderRadius: BorderRadius.circular(20),
      ),
      child: Text(
        text,

      ),

class _MedicationTile extends ConsumerWidget {
  final dynamic medication;
  const _MedicationTile({required this.medication});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    return Container(
      padding: const EdgeInsets.all(14),
      decoration: BoxDecoration(
        color: Colors.white,
        borderRadius: BorderRadius.circular(16),
      ),
      child: Row(
        children: [
          Container(
            width: 44,
            height: 44,
            decoration: BoxDecoration(
              color: AppTheme.accentBlue.withValues(alpha: 0.15),
              shape: BoxShape.circle,
            ),
            child: Icon(
              _iconForForm(medication.form),
              color: AppTheme.accentBlue,
            ),
          ),
          const SizedBox(width: 12),
          Expanded(
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Text(
                  medication.dosage == null
                      ? medication.name
                      : '${medication.name} ${medication.dosage}',
                  style: const TextStyle(
                    fontSize: 16,
                    fontWeight: FontWeight.w600,
                  ),
                ),
                const SizedBox(height: 2),
                Text(
                  _formatDateRange(medication.startDate, medication.endDate),
                  style: const TextStyle(fontSize: 13, color: Colors.black54),
                ),
              ],
            ),
          ),
          IconButton(
            icon: const Icon(Icons.delete_outline, color: Colors.red),
            onPressed: () => _confirmDelete(context, ref),
          ),
        ],
      ),
    );
  }

  Future<void> _confirmDelete(BuildContext context, WidgetRef ref) async {
    final ok = await showDialog<bool>(
      context: context,
      builder: (_) => AlertDialog(
        title: const Text('РЈРґР°Р»РёС‚СЊ Р»РµРєР°СЂСЃС‚РІРѕ?'),
        content: Text('В«${medication.name}В» Р±СѓРґРµС‚ СѓРґР°Р»РµРЅРѕ РІРјРµСЃС‚Рµ СЃ СЂР°СЃРїРёСЃР°РЅРёРµРј.'),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context, false),
            child: const Text('РћС‚РјРµРЅР°'),
          ),
          TextButton(
            onPressed: () => Navigator.pop(context, true),
            child: const Text('РЈРґР°Р»РёС‚СЊ', style: TextStyle(color: Colors.red)),
          ),
        ],
      ),
    );
    if (ok == true) {
      await ref.read(medicationsProvider.notifier).delete(medication.id!);
    }
  }

  IconData _iconForForm(MedicationForm form) {
    switch (form) {
      case MedicationForm.tablet:
        return Icons.medication;
      case MedicationForm.capsule:
        return Icons.medication_liquid;
      case MedicationForm.syrup:
        return Icons.sanitizer;
      case MedicationForm.injection:
        return Icons.vaccines;
      case MedicationForm.drops:
        return Icons.water_drop;
      case MedicationForm.other:
        return Icons.medical_services;
    }
  }

  String _formatDateRange(DateTime start, DateTime? end) {
    String fmt(DateTime d) => '${d.day}.${d.month}.${d.year}';
    if (end == null) return 'СЃ ${fmt(start)}';
    return '${fmt(start)} вЂ” ${fmt(end)}';
  }
}

class IntakeCard extends ConsumerWidget {
  final TodayIntake item;

  const IntakeCard({super.key, required this.item});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    return Container(
      padding: const EdgeInsets.symmetric(horizontal: 14, vertical: 12),
      decoration: BoxDecoration(
        color: AppTheme.cardWhite,
        borderRadius: BorderRadius.circular(16),
      ),
      child: Row(
        children: [
          // РРєРѕРЅРєР° С„РѕСЂРјС‹
          Container(
            width: 44,
            height: 44,
            decoration: BoxDecoration(
              color: _iconColor(item.medication.form).withValues(alpha: 0.15),
              shape: BoxShape.circle,
            ),
            child: Icon(
              _iconForForm(item.medication.form),
              color: _iconColor(item.medication.form),
              size: 22,
            ),
          ),

          const SizedBox(width: 12),

          // РќР°Р·РІР°РЅРёРµ + РІСЂРµРјСЏ
          Expanded(
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Text(
                  item.dosage == null
                      ? item.medicationName
                      : '${item.medicationName} ${item.dosage}',
                  style: const TextStyle(
                    fontSize: 16,
                    fontWeight: FontWeight.w600,
                  ),
                ),
                const SizedBox(height: 2),
                Text(
                  _formatTime(item.plannedAt),
                  style: const TextStyle(fontSize: 14, color: Colors.black54),
                ),
              ],
            ),
          ),

          // РљРЅРѕРїРєР° / СЃС‚Р°С‚СѓСЃ
          _buildStatusOrButton(context, ref),
        ],
      ),
    );
  }

  Widget _buildStatusOrButton(BuildContext context, WidgetRef ref) {
    switch (item.status) {
      case IntakeStatus.taken:
        return GestureDetector(
          onTap: () => _showUndoDialog(context, ref),
          child: const _StatusBadge(
            text: 'РџСЂРёРЅСЏС‚Рѕ',
            color: AppTheme.statusTaken,
          ),
        );
      case IntakeStatus.skipped:
        return const _StatusBadge(
          text: 'РџСЂРѕРїСѓС‰РµРЅРѕ',
          color: AppTheme.statusSkipped,
        );
      case IntakeStatus.postponed:
        return const _StatusBadge(
          text: 'РћС‚Р»РѕР¶РµРЅРѕ',
          color: AppTheme.statusPostponed,
        );
      case IntakeStatus.pending:
        return ElevatedButton(
          onPressed: () async {
            final repo = ref.read(intakeRepositoryProvider);
            await repo.markTaken(item.intake.id!, DateTime.now());

            // РћР±РЅРѕРІР»СЏРµРј РґР°РЅРЅС‹Рµ
            ref.invalidate(todayIntakesProvider);
            ref.invalidate(monthIntakesProvider);

            // РџРѕРєР°Р·С‹РІР°РµРј SnackBar СЃ РІРѕР·РјРѕР¶РЅРѕСЃС‚СЊСЋ РѕС‚РјРµРЅС‹
            if (context.mounted) {
              ScaffoldMessenger.of(context).showSnackBar(
                SnackBar(
                  content: const Text('Р›РµРєР°СЂСЃС‚РІРѕ РѕС‚РјРµС‡РµРЅРѕ РєР°Рє РїСЂРёРЅСЏС‚РѕРµ'),
                  action: SnackBarAction(
                    label: 'РћС‚РјРµРЅРёС‚СЊ',
                    onPressed: () async {
                      await repo.markPending(item.intake.id!);
                      ref.invalidate(todayIntakesProvider);
                      ref.invalidate(monthIntakesProvider);
                    },
                  ),
                ),
              );
            }
          },
          style: ElevatedButton.styleFrom(
            backgroundColor: AppTheme.statusTaken,
            foregroundColor: Colors.white,
            shape: RoundedRectangleBorder(
              borderRadius: BorderRadius.circular(20),
            ),
            padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 8),
          ),
          child: const Text('РџСЂРёРЅСЏР»', style: TextStyle(fontSize: 14)),
        );
    }
  }

  // Р”РёР°Р»РѕРіРѕРІРѕРµ РѕРєРЅРѕ РґР»СЏ РѕС‚РјРµРЅС‹ СЃР»СѓС‡Р°Р№РЅРѕРіРѕ РЅР°Р¶Р°С‚РёСЏ
  Future<void> _showUndoDialog(BuildContext context, WidgetRef ref) async {
    final ok = await showDialog<bool>(
      context: context,
      builder: (_) => AlertDialog(
        title: const Text('РћС‚РјРµРЅРёС‚СЊ РїСЂРёС‘Рј?'),
        content: const Text('Р’РµСЂРЅСѓС‚СЊ СЃС‚Р°С‚СѓСЃ РІ В«РћР¶РёРґР°РµС‚СЃСЏВ»?'),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context, false),
            child: const Text('РќРµС‚'),
          ),
          TextButton(
            onPressed: () => Navigator.pop(context, true),
            child: const Text('Р”Р°, РѕС‚РјРµРЅРёС‚СЊ'),
          ),
        ],
      ),
    );

    if (ok == true) {
      final repo = ref.read(intakeRepositoryProvider);
      await repo.markPending(item.intake.id!);
      ref.invalidate(todayIntakesProvider);
      ref.invalidate(monthIntakesProvider);
    }
  }

  IconData _iconForForm(MedicationForm form) {
    switch (form) {
      case MedicationForm.tablet:
        return Icons.medication;
      case MedicationForm.capsule:
        return Icons.medication_liquid;
      case MedicationForm.syrup:
        return Icons.sanitizer;
      case MedicationForm.injection:
        return Icons.vaccines;
      case MedicationForm.drops:
        return Icons.water_drop;
      case MedicationForm.other:
        return Icons.medical_services;
    }
  }

  Color _iconColor(MedicationForm form) {
    switch (form) {
      case MedicationForm.tablet:
        return const Color(0xFF3B82F6);
      case MedicationForm.capsule:
        return const Color(0xFFEF4444);
      case MedicationForm.syrup:
        return const Color(0xFFF59E0B);
      case MedicationForm.injection:
        return const Color(0xFF8B5CF6);
      case MedicationForm.drops:
        return const Color(0xFF10B981);
      case MedicationForm.other:
        return const Color(0xFF6B7280);
    }
  }

  String _formatTime(DateTime dt) {
    final h = dt.hour.toString().padLeft(2, '0');
    final m = dt.minute.toString().padLeft(2, '0');
    return '$h:$m';
  }
}

class _StatusBadge extends StatelessWidget {

}
